import http, { TOKEN_KEY } from "./http";
import ENDPOINTS from "./endpoints";

// Laravel peut renvoyer le contenu tel quel ou dans une clé "data".
function unwrap(body) {
  return body?.data ?? body;
}

function storeToken(data) {
  if (data?.token) {
    localStorage.setItem(TOKEN_KEY, data.token);
  }
}

const authService = {
  async login(credentials) {
    const response = await http.post(ENDPOINTS.auth.login, credentials);
    const data = unwrap(response.data);

    storeToken(data);

    return data;
  },

  async register(payload) {
    const response = await http.post(ENDPOINTS.auth.register, payload);
    const data = unwrap(response.data);

    storeToken(data);

    return data;
  },

  async my() {
    const response = await http.get(ENDPOINTS.auth.me);

    return unwrap(response.data);
  },

  async logout() {
    try {
      await http.post(ENDPOINTS.auth.logout);
    } finally {
      localStorage.removeItem(TOKEN_KEY);
    }
  },

  async updateProfile(payload) {
    const response = await http.put(ENDPOINTS.auth.updateProfile, payload);

    return unwrap(response.data);
  },
};

// Compatibilité avec les deux formes d'import.
export { authService };
export default authService;

import axios from "axios";

export const TOKEN_KEY = "alliance_token";
export const UNAUTHORIZED_EVENT = "auth:unauthorized";

const http = axios.create({
    baseURL: import.meta.env.VITE_API_BASE_URL || "http://127.0.0.1:8000/api/v1",
    headers: {
        Accept: "application/json",
        "Content-Type": "application/json",
    },
    timeout: 20000,
});

http.interceptors.request.use((config) => {
    const token = localStorage.getItem(TOKEN_KEY);

    if (token) {
        config.headers.Authorization = `Bearer ${token}`;
    }

    return config;
});

http.interceptors.response.use(
    (response) => response,
    (error) => {
        // Un 401 alors qu'un token existait : session expirée ou révoquée.
        // AuthContext écoute cet événement pour déconnecter l'interface.
        if (error.response?.status === 401 && localStorage.getItem(TOKEN_KEY)) {
            localStorage.removeItem(TOKEN_KEY);
            window.dispatchEvent(new Event(UNAUTHORIZED_EVENT));
        }

        return Promise.reject(error);
    }
);

export default http;
